	function withModalUtm(href) {
		var u;

		var arm = isDiscount() ? 'discount_modal' : 'retest_modal';
		// carry the visitor's own acquisition params (URL first, then the sp_utm cookie the passthrough script writes)

		var q = null, cookieUtm = {};
		try { q = new URLSearchParams(window.location.search); } catch (e) {}
		try { var m = document.cookie.match(/(?:^|; )sp_utm=([^;]*)/); if (m) cookieUtm = JSON.parse(decodeURIComponent(m[1])) || {}; } catch (e) {}
		for (var i = 0; i < KEYS.length; i++) {
			if (u.searchParams.has(KEYS[i])) continue;
			var v = q ? q.get(KEYS[i]) : null;
			if (v == null) v = cookieUtm[KEYS[i]];
			if (v != null && String(v).trim()) u.searchParams.set(KEYS[i], String(v).trim().slice(0, 512));
		}
		if (!u.searchParams.has('utm_source')) {
			u.searchParams.set('utm_source', 'superpower.com');
			u.searchParams.set('utm_medium', 'onsite_modal');
			u.searchParams.set('utm_campaign', 'always-on-modal');
		}
		if (!u.searchParams.has('utm_content')) u.searchParams.set('utm_content', arm);
		return u.pathname + u.search + u.hash;
	}

	// blur + opacity cross-fade between two state panels
	function transition(fromEl, toEl) {
		if (!toEl || fromEl === toEl) return;
		if (!fromEl) {
			toEl.classList.add('is-active', 'is-entering');
			toEl.addEventListener('animationend', function h() { toEl.removeEventListener('animationend', h); toEl.classList.remove('is-entering'); });
			return;
		}
		fromEl.classList.remove('is-entering');
		fromEl.classList.add('is-exiting');
		fromEl.addEventListener('animationend', function onExit() {
			fromEl.removeEventListener('animationend', onExit);
			fromEl.classList.remove('is-active', 'is-exiting');
			toEl.classList.add('is-active', 'is-entering');
			toEl.addEventListener('animationend', function onEnter() { toEl.removeEventListener('animationend', onEnter); toEl.classList.remove('is-entering'); });
		});
	}

				body: JSON.stringify({ data: { type: 'event', attributes: {
					metric: { data: { type: 'metric', attributes: { name: KLAVIYO_EVENT_METRIC } } },
					properties: { reason: selectedReason, reason_label: selectedLabel, image_variant: IMAGE_VARIANT, source: isDiscount() ? 'discount_modal' : 'retest_modal', offer: isDiscount() ? discountOfferKey() : FOLLOWUP_OFFER_KEY, promo_code: isDiscount() ? discountCode() : null },
					profile: { data: { type: 'profile', attributes: { email: email } } }
				} } })
			});
		} catch (e) {}
	}

	function handleSubmit() {
		if (!emailInput) return;
		hideFormError();
		var email = emailInput.value.trim();
		if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { showFormError('Please enter a valid email address'); emailInput.focus(); return; }
		setLoading(true);
		var params = new URLSearchParams(window.location.search);
		subscribeEmail(email, params)
			.then(function () {
				setLoading(false);
				hasSubmitted = true;
				try { localStorage.setItem(CLAIM_KEY, '1'); } catch (e) {}
				decorateCheckoutLinks();
				trackKlaviyoEvent(email);
				track('registration_started', { reason: selectedReason });
				transition(stateForm, stateSuccess);
			})
			.catch(function (err) { console.error('retest modal klaviyo error:', err); setLoading(false); showFormError('Something went wrong. Please try again.'); });
	}
	if (submitBtn) submitBtn.addEventListener('click', handleSubmit);
	if (emailInput) emailInput.addEventListener('keydown', function (e) { if (e.key === 'Enter') { e.preventDefault(); handleSubmit(); } });

	function scrollDepth() {
		var de = document.documentElement;
		var docH = Math.max(de.scrollHeight || 0, document.body ? document.body.scrollHeight || 0 : 0);
		var vh = window.innerHeight || de.clientHeight || 0;
		if (!docH || !vh || docH - vh < 100) return 0;
		return ((window.pageYOffset || de.scrollTop || 0) + vh) / docH;
	}
	var ticking = false;
	function onScroll() {
		if (ticking) return;
		ticking = true;
		requestAnimationFrame(function () {
			ticking = false;
			if (hasBeenShown) { detachScroll(); return; }
			// After a buy-box routing jump (sp3-cta-route) the depth was crossed by us, not the visitor.
			if (document.documentElement.hasAttribute('data-sp-offer-routed-at')) return;
			if (SCROLL_TRIGGER && scrollDepth() >= SCROLL_TRIGGER) { detachScroll(); show('scroll_depth'); }
		});
	}
	function detachScroll() { window.removeEventListener('scroll', onScroll); }
	// Time-on-page fallback: counts only while the tab is visible; retries while another overlay blocks the show.
	var visibleMs = 0;
	var lastTick = Date.now();
	var timeFallback = null;
	function stopTimeFallback() { if (timeFallback) { clearInterval(timeFallback); timeFallback = null; } }
	function startTimeFallback() {
		stopTimeFallback();
		if (!TIME_FALLBACK_MS) return;
		lastTick = Date.now();
		timeFallback = setInterval(function () {
			var now = Date.now();
			if (document.visibilityState !== 'hidden') visibleMs += now - lastTick;
			lastTick = now;
			if (hasBeenShown) { stopTimeFallback(); return; }
			if (visibleMs >= TIME_FALLBACK_MS) { detachScroll(); show('time_on_page'); }
			if (visibleMs >= TIME_FALLBACK_MS * 6) stopTimeFallback();
		}, 1000);
	}

	// debug handles
	window.__spModal = {
		config: function () { return { testPage: isTestPage, scrollTrigger: SCROLL_TRIGGER, timeFallbackMs: TIME_FALLBACK_MS }; },
		show: function () { hasBeenShown = false; refreshOffer(); offerLocked = true; if (isDiscount()) applyDiscountCopy(); overlay.classList.add('is-visible'); lockScroll(); },
		offer: function () { return currentOffer; },
		image: function () { return IMAGE_VARIANT; },
		hide: hide,
		reset: function () {
			try { localStorage.removeItem(SESSION_KEY); } catch (e) {}
			try { sessionStorage.removeItem(SESSION_KEY); } catch (e) {}
			hasBeenShown = false; hide();
			visibleMs = 0;
			window.addEventListener('scroll', onScroll, { passive: true });
			startTimeFallback();
		},
		depth: scrollDepth
	};

	window.addEventListener('pageshow', auditLock);
	window.addEventListener('popstate', auditLock);
	document.addEventListener('visibilitychange', auditLock);
	window.addEventListener('focus', auditLock);
	setInterval(auditLock, 1000);
	auditLock();

	var strayChecks = 0;
	var strayWatch = setInterval(function () {
		strayChecks++;
		if (strayChecks > 120) { clearInterval(strayWatch); return; }
		var de = document.documentElement;
		if (de.getAttribute(LOCK_ATTR) === '1') return;
		if (de.style.overflow === 'hidden' && !someoneOwnsLock()) {
			de.style.overflow = '';
			if (document.body) document.body.style.overflow = '';
		}
	}, 1000);

	if (recentlyShown()) return;
	if (beforeYouGoShownThisSession()) return;
	window.addEventListener('scroll', onScroll, { passive: true });
	startTimeFallback();
})();
